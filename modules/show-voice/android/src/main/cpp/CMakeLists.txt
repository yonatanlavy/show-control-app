cmake_minimum_required(VERSION 3.22.1)
project(showvoice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(WHISPER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/whisper.cpp")
if(NOT EXISTS "${WHISPER_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Run node scripts/prepare-whisper.mjs before building ShowVoice")
endif()
add_subdirectory("${WHISPER_SOURCE}" whisper-build)
add_library(showvoice SHARED VoiceJni.cpp ../../../../shared/WhisperBridge.cpp)
target_include_directories(showvoice PRIVATE ../../../../shared "${WHISPER_SOURCE}/include")
target_link_libraries(showvoice PRIVATE whisper android log)
